import numpy as np
import matplotlib.pyplot as plt

# 1. 设置参数 t 的范围 (0 到 2π)
t = np.linspace(0, 2 * np.pi, 1000)

# 2. 定义心脏线方程 (修改数字可调整形状)
x =  * np.sin(t)**3
y =  * np.cos(t) -  * np.cos(2*t) -  * np.cos(3*t) -  * np.cos(4*t)

# 3. 创建画布并绘图
plt.figure(figsize=(8, 6))
plt.plot(x, y, color='#ef4444', linewidth=2)

# 4. 设置样式与标题
plt.title('Heart Curve')
plt.axis('equal')  # 保持比例一致
plt.axis('off')    # 隐藏坐标轴

# 5. 显示结果
plt.show()
